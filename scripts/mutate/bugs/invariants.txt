# The invariants the table is built on, each one broken on purpose, and what notices.
#
# A sweep asks "would anything notice this token changing". This asks the question the other way
# round: here is a mistake a person would plausibly make in this code, does the suite say so. The
# two find different things -- a sweep cannot express "pop_back before repointing instead of
# after", and a person would not think to try "2 -> 3" in a distance comparison.
#
# Grouped by the invariant being broken rather than by function, because that is what a survivor
# here means: not "this line is untested" but "this promise is untested".
#
#     scripts/mutate/mutate.py --bugs scripts/mutate/bugs/invariants.txt
#
# A snapshot against the code as it was: when a block stops applying, that part has been rewritten
# and the question wants re-deriving rather than repairing.

# ---------------------------------------------------------------- probing and the group index

# the home bucket comes from the low bits of the hash rather than the high ones
# m_shifts and m_bucket_mask describe the same array two ways, and only the shift is applied to a
# first probe -- do_find_hashed indexes with hash >> m_shifts and does not mask. Taking the low
# bits still lands inside the array, so this is not a crash; it is every key whose low bits collide
# piling into one run.
<<<
        return static_cast<value_idx_type>(hash >> m_shifts);
===
        return static_cast<value_idx_type>(hash) & m_bucket_mask;
>>>

# the probe walks off the end of the group array instead of wrapping
# m_group_mask is the number of groups minus one, so masking is the wrap. Without it the walk runs
# past the array as soon as a probe starts near the end, which is a read out of bounds long before
# it is a wrong answer -- so a plain build may well stay green where a sanitizer build screams.
<<<
        return static_cast<value_idx_type>((group_idx + (++delta)) & m_group_mask);
===
        return static_cast<value_idx_type>(group_idx + (++delta));
>>>

# the probe stays on the same group forever instead of stepping through the sequence
# delta is what makes the sequence triangular, and incrementing it is what makes it move at all.
# Read without the increment the first step is zero, so a probe that has to leave its home group
# never leaves it: a miss in a full group does not terminate.
<<<
        return static_cast<value_idx_type>((group_idx + (++delta)) & m_group_mask);
===
        return static_cast<value_idx_type>((group_idx + delta) & m_group_mask);
>>>

# a fingerprint of zero is stored as zero, which is the marker for "empty slot"
# The low byte of a hash is zero for one key in 256, and mapping that to 8 is the whole reason an
# empty slot can be told from an occupied one by the fingerprint alone. Without the remap such a
# key is stored in a slot that every later probe reads as free: the key cannot be found, and the
# next insert overwrites its slot.
<<<
        t[i] = (i == 0 ? 8U : i) * 0x01010101U;
===
        t[i] = i * 0x01010101U;
>>>

# an insert that leaves a full group behind does not record that it did
# The counter is the only thing that tells a later probe to look past this group. Without the
# increment the probe stops at the home group, and every key that had to move on is unfindable --
# but only once a group actually fills, so a small table stays green.
<<<
            if (group.m_overflows[counter] != 255) {
                ++group.m_overflows[counter];
            }
===
            if (group.m_overflows[counter] == 255) {
                ++group.m_overflows[counter];
            }
>>>

# an erase undoes the counters from the group the entry landed in rather than from its home
# The walk has to retrace exactly the groups the insert passed. Starting it at the end retraces
# nothing, so the counters only ever grow. Before the probe was bounded this hung -- every group
# left with a stale count for a class, and a miss with nothing to stop at -- so the sibling bug in
# bugs/erase-path.txt was caught as a hang. The bound stops the miss at the end of the array
# instead, so this now SURVIVES: the counters being too high only lengthens probes, which no answer
# shows. It is the shape of bug the suite is least able to see, kept as a named reminder of that.
<<<
        auto group_idx = group_idx_from_hash(mh);
        if (group_idx != found_in) {
===
        auto group_idx = found_in;
        if (group_idx != found_in) {
>>>

# the erase clears the fingerprint of the wrong slot in the group
# slot is a slot index over the whole array; the lane inside its group is slot % 16 and the group
# is slot / 16. Swapping them clears some other slot's fingerprint, which loses that key and leaves
# the erased one findable.
<<<
        groups[found_in].m_fingerprints[slot % slots_per_group] = 0;
===
        groups[found_in].m_fingerprints[slot / 16] = 0;
>>>

# a probe trusts the fingerprint and never compares the key
# One byte of hash matches by accident once in 256, so this is wrong from the first collision and
# not before: a lookup for an absent key comes back with somebody else's value.
<<<
                auto const value_idx = index[slot];
                if (m_equal(key, get_key(m_values[value_idx]))) {
                    return {slot, value_idx, true};
                }
===
                auto const value_idx = index[slot];
                if (true) {
                    return {slot, value_idx, true};
                }
>>>

# the candidate loop drops the lowest bit by shifting instead of clearing it
# `lanes &= lanes - 1` clears the lowest set bit; a shift moves every remaining candidate down one
# lane as well, so after the first mismatch every further candidate is looked up in the wrong slot.
<<<
                lanes &= lanes - 1;
            }
            // Not here if nothing of this class ever overflowed past this group, and not anywhere
===
                lanes >>= 1;
            }
            // Not here if nothing of this class ever overflowed past this group, and not anywhere
>>>

# the search for the slot that points at a value accepts any fingerprint match
# slot_of_value is how an erase by iterator and the dense move find the slot of a *known* value.
# The fingerprint narrows the candidates; only the index identifies the one. Without that check the
# first key sharing a fingerprint is repointed instead, which corrupts two entries at once.
<<<
                if (index[slot] == value_idx) {
                    return slot;
                }
===
                if (true) {
                    return slot;
                }
>>>

# empty slot that no longer exists.
<<<

        return size() > m_max_bucket_capacity;
    }

# the load factor limit is taken as a count of buckets rather than a fraction of them
# m_max_bucket_capacity would exceed bucket_count(), is_full() would never fire, and the table
# fills completely -- the same end state as never growing, reached a different way.
<<<

        }
    }
===
            m_max_bucket_capacity = static_cast<value_idx_type>(num_buckets);
        }
    }
>>>

# max_load_factor accepts a value above 1
# The comment on this line says what happens: the capacity exceeds the bucket count, is_full() never
# fires, and place_and_shift_up probes forever for an empty bucket that does not exist.
<<<
        m_max_load_factor = (std::min)(ml, 1.0F);
===
        m_max_load_factor = ml;
>>>

# calc_shifts_for_size sizes the array for one element too few
# The table is then born exactly one insert short of needing to grow, which is only visible if
# something reserves and then inserts precisely that many.
<<<

               static_cast<std::size_t>(static_cast<float>(calc_num_buckets(shifts)) * max_load_factor()) < s) {
===

>>>

# reserve grows the values but not the buckets
# So reserve() stops being a promise about lookups and becomes one about m_values only; the table
# still rehashes on the way up, just later than asked.
<<<

        if (0 == bucket_count() || shifts < m_shifts) {

            fill_buckets_from_values();

        if (0 == bucket_count()) {
            allocate_buckets_from_shift(shifts);

>>>

# rehash resizes the bucket array and leaves it empty
# Same shape as the growth bug above, reached through the public API instead.
<<<
        if (shifts != m_shifts) {
            allocate_buckets_from_shift(shifts);

# the group mask is the number of groups rather than one less than it
# Masking with the count instead of count-1 lets a probe address one group past the end, and stops
# masking off the top bit that a hash puts there: the wrap is wrong and the read is out of bounds.
<<<
        m_group_mask = static_cast<value_idx_type>(num_groups - 1);
===
        m_group_mask = static_cast<value_idx_type>(num_groups);

# ---------------------------------------------------------------- hashing

# a hash that never said it avalanches is used as-is
# wyhash is what turns std::hash<int>'s identity function into something with entropy in the high
# bits -- which is where group_idx_from_hash takes the index from. Without it every small integer
# key lands in the same handful of buckets.
<<<

===

>>>

# a 32 bit avalanching hash is used without being spread into the high bits
# The multiply is what moves 32 bits of entropy up to where the shift reads them. Without it every
